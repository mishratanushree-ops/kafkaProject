import org.apache.kafka.clients.producer.KafkaProducer;
import org.apache.kafka.clients.producer.ProducerConfig;
import org.apache.kafka.clients.producer.ProducerRecord;
import org.apache.kafka.clients.producer.RecordMetadata;
import org.apache.kafka.common.serialization.StringSerializer;
import java.util.Properties;
import java.util.concurrent.Future;

public class KeyBasedPartitioningExample {

    public static void main(String[] args) {
        // 1. Configure the Producer
        Properties properties = new Properties();
        properties.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
        properties.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, StringSerializer.class.getName());
        properties.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, StringSerializer.class.getName());

        // 2. Instantiate the KafkaProducer
        KafkaProducer<String, String> producer = new KafkaProducer<>(properties);
        String topic = "order-events";

        try {
            // Simulated sequence of events for two different orders
            // order-123 and order-456 will likely land on different partitions, 
            // but all order-123 events will land on the exact same partition.
            sendEvent(producer, topic, "order-123", "Created");
            sendEvent(producer, topic, "order-456", "Created");
            sendEvent(producer, topic, "order-123", "Paid");
            sendEvent(producer, topic, "order-456", "Shipped");
            sendEvent(producer, topic, "order-123", "Shipped");

        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            // 4. Close the producer to flush pending logs
            producer.close();
        }
    }

    private static void sendEvent(KafkaProducer<String, String> producer, String topic, String key, String value) throws Exception {
        // 3. Create the Record with a KEY to enforce partitioning
        ProducerRecord<String, String> record = new ProducerRecord<>(topic, key, value);

        // Send asynchronously and print partition details on acknowledgment
        Future<RecordMetadata> future = producer.send(record, (metadata, exception) -> {
            if (exception == null) {
                System.out.printf("Key: %s | Value: %-8s -> Partition: %d | Offset: %d%n",
                        key, value, metadata.partition(), metadata.offset());
            } else {
                System.err.println("Error producing record: " + exception.getMessage());
            }
        });
        
        // Block briefly just to keep console output chronological for this demo
        future.get();
    }
}

Crucial Architectural RulesFixed Partition Counts: Key-based routing relies on the mathematical formula hash(key) % total_partitions. If you dynamically scale up or alter the number of partitions in a live topic, subsequent hashes will resolve to completely different partitions, breaking your historic ordering guarantees

import org.apache.kafka.clients.consumer.ConsumerConfig;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.apache.kafka.clients.consumer.ConsumerRecords;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.apache.kafka.common.serialization.StringDeserializer;

import java.time.Duration;
import java.util.Collections;
import java.util.Properties;

public class KeyValueConsumerExample {

    public static void main(String[] args) {
        // 1. Configure the Consumer properties
        Properties properties = new Properties();
        properties.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
        properties.put(ConsumerConfig.KEY_DESERIALIZER_CLASS_CONFIG, StringDeserializer.class.getName());
        properties.put(ConsumerConfig.VALUE_DESERIALIZER_CLASS_CONFIG, StringDeserializer.class.getName());
        
        // Consumer group splits partitions among members; items in the same group share work
        properties.put(ConsumerConfig.GROUP_ID_CONFIG, "order-processing-group");
        
        // Start reading from the earliest uncommitted message
        properties.put(ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, "earliest");

        // 2. Create the KafkaConsumer instance
        KafkaConsumer<String, String> consumer = new KafkaConsumer<>(properties);

        // 3. Subscribe to the target topic
        String topic = "order-events";
        consumer.subscribe(Collections.singletonList(topic));

        System.out.println("Listening for ordered order-events... Press Ctrl+C to exit.");

        // Clean shutdown hook
        Thread mainThread = Thread.currentThread();
        Runtime.getRuntime().addShutdownHook(new Thread(() -> {
            System.out.println("\nDetected shutdown. Waking up consumer...");
            consumer.wakeup(); // Thread-safe method to interrupt the consumer poll
            try {
                mainThread.join();
            } catch (InterruptedException e) {
                e.printStackTrace();
            }
        }));

        try {
            // 4. Poll and process data in an infinite loop
            while (true) {
                // Poll Kafka for new data records (waiting up to 100ms if empty)
                ConsumerRecords<String, String> records = consumer.poll(Duration.ofMillis(100));

                for (ConsumerRecord<String, String> record : records) {
                    // Extracting the key, value, partition, and offset information
                    String key = record.key();
                    String value = record.value();
                    int partition = record.partition();
                    long offset = record.offset();

                    // Business Logic processing guarantees chronological execution per key
                    System.out.printf("Processing -> Partition: %d | Offset: %d | Key: %s | Status update: %s%n",
                            partition, offset, key, value);
                }
            }
        } catch (org.apache.kafka.common.errors.WakeupException e) {
            System.out.println("Consumer received wakeup signal.");
        } catch (Exception e) {
            System.err.println("Unexpected error occurred: " + e.getMessage());
        } finally {
            // 5. Close consumer and commit offsets explicitly/safely before exit
            consumer.close();
            System.out.println("Consumer safely closed.");
        }
    }
}

Critical Consumer Rules for OrderingSingle-Threaded Processing Per Partition: A single partition can never be read by more than one consumer thread in the same group.id. This rule prevents race conditions between your events.Do Not Use Multi-Threaded Workers Carelessly: If your consumer thread receives records and blindly dispatches them to an internal Java ExecutorService thread pool, you will immediately lose your order guarantee. The records will execute out of order across your CPU threads.Scale by Partition Count: If you need more consumer throughput, you must increase the number of partitions on the topic. Kafka will then rebalance and assign those new partitions to your additional application instances.
